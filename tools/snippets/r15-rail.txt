 *
 * 48 卷 L55 逐字顺序：「首页 / 时间轴 / 交接账本 / 记忆库 / 召回审查 / 团队 / 日历 / 设置」。
 * 48 卷 L57：「第 2 项（时间轴）为选中态」⇒ **默认 activeIndex = 1**。
 * 42 卷 Q2 用户原话「它肯定是页签内自带」⇒ 本表是**页签内左栏**的数据源，不新增宿主 view。
 * ★纯数据 + 纯函数，零 IO、零 DOM、确定性（与 R12–R14 同款纪律）。
 */
export const WB_RAIL_ITEMS_PRE_V1 = [
  { key: 'home', label: '首页' },
  { key: 'timeline', label: '时间轴' },
  { key: 'ledger', label: '交接账本' },
  { key: 'library', label: '记忆库' },
  { key: 'recall', label: '召回审查' },
  { key: 'team', label: '团队' },
  { key: 'calendar', label: '日历' },
  { key: 'settings', label: '设置' },
]

/**
 * 15a：左栏结构。返回 8 项（key/label/order/active）+ activeIndex/activeKey。
 * 非法或未知 active ⇒ **回落 index 1（时间轴）**（48 卷 L57），不抛。
 */
export function buildRailPre(opts = {}) {
  const o = (opts && typeof opts === 'object') ? opts : {}
  const want = typeof o.active === 'string' ? o.active.trim() : ''
  let idx = 1
  if (want) {
    const found = WB_RAIL_ITEMS_PRE_V1.findIndex((it) => it.key === want)
    if (found >= 0) idx = found
  }
  const items = WB_RAIL_ITEMS_PRE_V1.map((it, i) => ({ key: it.key, label: it.label, order: i, active: i === idx }))
  return { items, total: items.length, activeIndex: idx, activeKey: items[idx].key }
}

/**
 * 15b：与泳道/主区视图联动 —— 左栏 key → 主区视图模式（纯映射）。
 * 未知 key ⇒ 'timeline'（默认态，与 buildRailPre 的默认一致）。
 */
export function railToViewPre(railKey) {
  const k = typeof railKey === 'string' ? railKey.trim() : ''
  const map = {
    home: 'home',
    timeline: 'timeline',
    ledger: 'ledger',
    library: 'library',
    recall: 'recall',
    team: 'team',
    calendar: 'calendar',
    settings: 'settings',
  }
  return Object.prototype.hasOwnProperty.call(map, k) ? map[k] : 'timeline'
}
